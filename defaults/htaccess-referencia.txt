# sigetec.cl — Astro estático en cPanel (Apache)
Options -Indexes
DirectoryIndex index.html

<IfModule mod_rewrite.c>
  RewriteEngine On
  # Formulario de contacto: /api/contacto -> contacto.php
  RewriteRule ^api/contacto/?$ /api/contacto.php [L,QSA]
  # Forzar HTTPS
  RewriteCond %{HTTPS} !=on
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]
  # Redirecciones desde el sitio WordPress anterior (agregar las que falten)
  RedirectMatch 301 ^/blog/?$ /casos/
  RedirectMatch 301 ^/category/.*$ /casos/
  RedirectMatch 301 ^/tienda/?.*$ /productos/
  RedirectMatch 301 ^/shop/?.*$ /productos/
  RedirectMatch 301 ^/contacto-2/?$ /contacto/
  RedirectMatch 301 ^/quienes-somos/?$ /nosotros/
  RedirectMatch 301 ^/servicios/?$ /soluciones/
</IfModule>

ErrorDocument 404 /404.html

<IfModule mod_headers.c>
  Header set X-Content-Type-Options "nosniff"
  Header set X-Frame-Options "SAMEORIGIN"
  Header set Referrer-Policy "strict-origin-when-cross-origin"
  Header set Permissions-Policy "camera=(), microphone=(), geolocation=()"
  <FilesMatch "\.(woff2|jpg|jpeg|png|ico|svg|webp|css|js|wasm|pf_index|pf_fragment|pf_meta|pagefind)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.(html|xml|webmanifest|txt)$">
    Header set Cache-Control "public, max-age=600"
  </FilesMatch>
  # Parámetros editables (Google Analytics): no cachear por un año
  <FilesMatch "^site-config\.js$">
    Header set Cache-Control "public, max-age=300"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json application/manifest+json image/svg+xml text/plain application/xml
</IfModule>

# Tipos MIME que algunos servidores no traen de fábrica
AddType font/woff2 .woff2
AddType application/manifest+json .webmanifest
AddType application/wasm .wasm
AddType application/octet-stream .pf_index .pf_fragment .pf_meta .pagefind
AddType image/webp .webp
AddDefaultCharset UTF-8

# Proteger la configuración del formulario
<Files "config.php">
  Require all denied
</Files>
